# .gitignore

Настройки
```bash
----------Игнорирование по имени файла------------
# Игнорировать файл todo.txt в любой папке
todo.txt

# Игнорировать файл todo.txt только в корне репозитория
/todo.txt

# Игнорировать все файлы .log в любой папке
*.log

# Игнорировать все файлы, у которых есть два любых знака после слова file и окончание .txt
file??.txt

# Игнорировать все файлы, начинающиеся на temp
temp*

# Игнорировать все файлы, заканчивающиеся на .tmp
*.tmp

# Игнорировать все файлы, содержащие слово backup
*backup*

# Игнорировать файлы с расширениями .pyc, .pyo, .pyd
*.py[cod]

# Игнорировать файлы с расширением из одного символа (a.js, b.css)
?.*

----------Игнорирование по имени папки------------
# Игнорировать папку build в любой папке
build/

# Игнорировать папку build только в корне репозитория
/build/

# Игнорировать папку node_modules в любой папке
node_modules/

# Игнорировать все папки, начинающиеся на temp
temp*/

# Игнорировать все папки, заканчивающиеся на cache
*cache/

# Игнорировать папку с именем bin, независимо от регистра
[Bb]in/

# Игнорировать папку target в любой папке
target/

# Игнорировать папку .idea в любой папке
.idea/

----------Игнорирование по пути------------
# Игнорировать файл config.json только в папке src
src/config.json

# Игнорировать папку logs только внутри папки app
app/logs/

# Игнорировать все файлы .txt внутри папки docs на любом уровне
docs/**/*.txt

# Игнорировать всё содержимое папки dist
dist/**

# Игнорировать папку build только на верхнем уровне проекта
/build/

----------Игнорирование по шаблону------------
# Игнорировать все файлы, начинающиеся на test_
test_*

# Игнорировать все файлы, заканчивающиеся на _backup
*_backup

# Игнорировать все файлы, содержащие слово draft
*draft*

# Игнорировать все файлы с расширением .bak в любой папке
*.bak

# Игнорировать все файлы, состоящие из одного символа и расширения
?.*

# Игнорировать все файлы, у которых ровно три символа перед .txt
???.txt

# Игнорировать все файлы, у которых от одного до трёх символов перед .txt
[?][?][?].txt

----------Исключения из игнорирования------------
# Не игнорировать файл important.log, даже если все .log игнорируются
!important.log

# Не игнорировать папку build внутри папки src
!src/build/

# Не игнорировать файл .env.example, даже если все .env игнорируются
!.env.example

# Не игнорировать папку .vscode, но игнорировать всё внутри
.vscode/*
!.vscode/settings.json
!.vscode/launch.json

----------Комбинированные правила------------
# Игнорировать все .log, кроме important.log
*.log
!important.log

# Игнорировать всё в папке logs, кроме файла keep.log
logs/*
!logs/keep.log

# Игнорировать все .txt, кроме файлов в папке docs
*.txt
!docs/**/*.txt

# Игнорировать папку temp везде, кроме корня проекта
temp/
!/temp/

# Игнорировать все файлы .class, но не в папке src
*.class
!src/**/*.class
```
